# PnE (Plan-and-Execute) 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** COMPLEX 路径内部二级分流：简单复杂走现有 ReAct（零改动），真正多步走 Planner → Executor → Evaluator → Synthesizer，支持并行/串行拓扑调度、提前终止、replan（上限 2 次）。

**Architecture:** PnE 为独立模块 `server/src/services/pne/`，生成器函数编排（AsyncGenerator\<AgentEvent\>），复用现有 SSE 管道。Executor 用 ChatOpenAI.bindTools 手写 mini 循环（不经过 langchain-adapter，无 pendingToolCalls 串扰）。计划快照经 `plan` SSE 事件写入 thought_steps 持久化，前端渲染 todo list。

**Tech Stack:** LangChain ChatOpenAI（tool-calling）、callLLM/streamLLM（llm-caller.ts）、Vitest、Vue 3 + Pinia。

**Spec:** `docs/superpowers/specs/2026-10-09-pne-design.md`（四项已澄清决定 D1-D4 在此文档，冲突时以 spec 为准）

```typescript
// server/src/services/pne/types.ts
export type StageStatus = 'pending' | 'running' | 'done' | 'skipped'
export type StageStrategy = 'tool' | 'reasoning' | 'hybrid'

export interface Stage {
  index: number
  goal: string
  strategy: StageStrategy
  toolHint?: string
  toolArgs?: string
  dependsOn: number[]
}

export interface ExecutionPlan {
  reasoning: string
  stages: Stage[]
}

export interface StageResult {
  index: number
  goal: string
  outcome: string
}

/** 与 llm-caller 的 ChatMessage 结构一致（该类型未导出，此处镜像定义） */
export interface PneChatMessage {
  role: 'user' | 'assistant' | 'system'
  content: string
}

export const MAX_STAGES = 5

/** 解析并防御性归一 Planner 输出。非法结构抛错，由上层回退 ReAct。 */
export function parsePlan(json: string): ExecutionPlan {
  const data = JSON.parse(json) as { reasoning?: unknown; stages?: unknown }
  if (!Array.isArray(data.stages) || data.stages.length === 0) {
    throw new Error('计划缺少有效 stages')
  }
  if (data.stages.length > MAX_STAGES) {
    throw new Error(`阶段数 ${data.stages.length} 超上限 ${MAX_STAGES}`)
  }
  const stages: Stage[] = data.stages.map((s, i) => {
    const st = s as Record<string, unknown>
    if (typeof st.goal !== 'string' || !st.goal.trim()) {
      throw new Error(`阶段 ${i + 1} 缺 goal`)
    }
    const index = typeof st.index === 'number' ? st.index : i + 1
    const strategy: StageStrategy =
      st.strategy === 'reasoning' || st.strategy === 'hybrid' ? st.strategy : 'tool'
    const dependsOn = Array.isArray(st.dependsOn)
      ? st.dependsOn.filter((d): d is number => typeof d === 'number' && d !== index)
      : []
    return {
      index,
      goal: st.goal,
      strategy,
      toolHint: typeof st.toolHint === 'string' && st.toolHint.trim() ? st.toolHint : undefined,
      toolArgs: typeof st.toolArgs === 'string' && st.toolArgs.trim() ? st.toolArgs : undefined,
      dependsOn,
    }
  })
  // 依赖指向不存在的阶段 -> 丢弃，防死锁
  const indexes = new Set(stages.map(s => s.index))
  for (const s of stages) {
    s.dependsOn = s.dependsOn.filter(d => indexes.has(d))
  }
  return { reasoning: typeof data.reasoning === 'string' ? data.reasoning : '', stages }
}

/** 阶段结果格式化为文本块（Planner/Evaluator/Executor/Synthesizer 共用） */
export function formatStageResults(results: StageResult[]): string {
  if (results.length === 0) return '（尚无已完成阶段）'
  return results
    .map(r => `阶段 ${r.index}. ${r.goal}\n结果：${r.outcome}`)
    .join('\n\n')
}
```

```typescript
// server/src/services/pne/gate.ts
import { callLLM } from '../llm-caller'
import { MODEL_LIGHT, type Complexity } from '../llm-config'

/** 规划信号词（与 ARCHITECTURE.md §3.1 判据表一致） */
const PNE_SIGNAL = /(对比|分析|方案|规划|步骤|制定|策划|评估)/
const PNE_LENGTH_THRESHOLD = 120

const PNE_LLM_PROMPT =
  '判断该问题是否需要多阶段（规划→分步执行→汇总）处理。多阶段问题如：多条件对比分析、多信息源汇总、方案制定。单一信息可答的问题输出 no。只输出 yes 或 no。'

/** 纯规则判据：deep 强制 PnE；长问题或含规划信号词走 PnE */
export function shouldUsePneByRules(query: string, complexity: Complexity): boolean {
  if (complexity === 'deep') return true
  if (query.length > PNE_LENGTH_THRESHOLD) return true
  return PNE_SIGNAL.test(query)
}

/** 完整判据：规则优先，未命中时 LLM 轻判兜底。LLM 失败视为不需要（回退 ReAct 更安全）。 */
export async function shouldUsePne(query: string, complexity: Complexity): Promise<boolean> {
  if (shouldUsePneByRules(query, complexity)) return true
  try {
    const result = await callLLM(
      [{ role: 'user', content: `用户问题: "${query}"` }],
      PNE_LLM_PROMPT,
      MODEL_LIGHT,
      20
    )
    return /yes/i.test(result)
  } catch {
    return false
  }
}
```

```typescript
const PLANNER_MAX_TOKENS = 2000

/** 把对话历史压成文本（取最近 6 条，防 prompt 膨胀） */
function formatHistory(history: PneChatMessage[]): string {
  const recent = history.slice(-6)
  if (recent.length === 0) return '（无）'
  return recent.map(m => `${m.role}: ${m.content.slice(0, 300)}`).join('\n')
}

 * Planner：query + 已完成阶段结果 → ExecutionPlan。
 * 解析失败抛错，由 runPneAgent 上抛触发 ReAct 回退。
 */
export async function* planQuery(
  query: string,
  stageResults: StageResult[],
  history: PneChatMessage[],
  toolList: string
): AsyncGenerator<AgentEvent, ExecutionPlan> {
  yield { type: 'thought', content: stageResults.length === 0 ? '【规划】正在拆解任务...' : '【规划】重新规划中...' }

  const system = renderPrompt(loadPrompt('pne/planner'), {
    toolList: toolList || '（无可用工具）',
    dateContext: buildDateContext(),
    stageResults: formatStageResults(stageResults),
    history: formatHistory(history),
    query,
  })

  const raw = await callLLM(
    [{ role: 'user', content: `用户问题：${query}` }],
    system,
    MODEL_STRONG,
    PLANNER_MAX_TOKENS
  )

  const json = extractFirstJsonObject(stripMarkdownCodeFence(raw))
  if (!json) throw new Error(`Planner 输出中未找到 JSON: ${raw.slice(0, 100)}`)
  return parsePlan(json)
}
```

```typescript
// server/src/services/pne/evaluator.ts
import { callLLM } from '../llm-caller'
import { MODEL_LIGHT } from '../llm-config'
import { loadPrompt, renderPrompt } from '../prompt-loader'
import { formatStageResults, type ExecutionPlan, type StageResult } from './types'

export type PneDecision = 'continue' | 'achieved' | 'replan' | 'stuck'

const DECISIONS: readonly PneDecision[] = ['continue', 'achieved', 'replan', 'stuck']

function formatPlanStages(plan: ExecutionPlan): string {
  return plan.stages
    .map(s => `${s.index}. ${s.goal} [${s.strategy}${s.dependsOn.length ? ` 依赖${s.dependsOn.join(',')}` : ''}]`)
    .join('\n')
}

/** Evaluator：判定 continue/achieved/replan/stuck。输出非法抛错，由上层回退。 */
export async function evaluateProgress(
  query: string,
  plan: ExecutionPlan,
  stageResults: StageResult[]
): Promise<{ decision: PneDecision; reason: string }> {
  const system = renderPrompt(loadPrompt('pne/evaluator'), {
    query,
    planStages: formatPlanStages(plan),
    stageResults: formatStageResults(stageResults),
  })
  const raw = await callLLM(
    [{ role: 'user', content: '请评估当前执行状态。' }],
    system,
    MODEL_LIGHT,
    300
  )

  const line = raw
    .split('\n')
    .map(l => l.trim())
    .find(l => DECISIONS.some(d => l.toLowerCase().startsWith(d)))
  if (!line) throw new Error(`Evaluator 输出非法: ${raw.slice(0, 100)}`)

  const [word, ...rest] = line.split('|')
  const decision = word.trim().toLowerCase()
  if (!DECISIONS.includes(decision as PneDecision)) {
    throw new Error(`Evaluator 判定词非法: ${line.slice(0, 100)}`)
  }
  return { decision: decision as PneDecision, reason: rest.join('|').trim() }
}
```

```typescript
// server/src/services/pne/executor.ts
import type { AgentEvent } from '../../types'
import type { DynamicStructuredTool } from '@langchain/core/tools'
import { AIMessage, SystemMessage, HumanMessage, ToolMessage } from '@langchain/core/messages'
import type { BaseMessage } from '@langchain/core/messages'
import type { ToolCall } from '@langchain/core/messages/tool'
import { createChatOpenAI, MODEL_STRONG } from '../llm-config'
import { loadPrompt, renderPrompt } from '../prompt-loader'
import { buildDateContext, buildKnowledgeContext } from '../knowledge'
import { logToolCall } from '../logger'
import { formatStageResults, type Stage, type StageResult } from './types'

const MAX_TOOL_ROUNDS = 3
const MAX_CONSECUTIVE_FAILURES = 3
const OBSERVATION_TRUNCATE = 4000

export interface ExecutorDeps {
  tools: DynamicStructuredTool<Record<string, unknown>>[]
  stageResults: StageResult[]
  query: string
  userId?: string
  conversationId?: string
}

function isToolFailure(output: string): boolean {
  return output.startsWith('Tool error:')
}

/**
 * Executor：单阶段 mini 循环（不经过 langchain-adapter，无 pendingToolCalls 串扰）。
 * LLM 发 tool_calls 就执行工具并回注，最多 MAX_TOOL_ROUNDS 轮；
 * 不发 tool_calls 即阶段完成。轮次耗尽/连续失败时追加一次总结调用。
 */
export async function executeStage(
  stage: Stage,
  deps: ExecutorDeps
): Promise<{ events: AgentEvent[]; outcome: string }> {
  const events: AgentEvent[] = []
  const toolMap = new Map(deps.tools.map(t => [t.name, t]))
  const toolList = deps.tools.map(t => `- ${t.name} - ${t.description}`).join('\n')

  const system = renderPrompt(loadPrompt('pne/executor'), {
    stageGoal: stage.goal,
    strategy: stage.strategy,
    toolHint: stage.toolHint ?? '无',
    toolArgs: stage.toolArgs ?? '无',
    toolList: toolList || '（无可用工具）',
    stageResults: formatStageResults(deps.stageResults),
    dateContext: buildDateContext(),
    knowledgeContext: buildKnowledgeContext(),
    query: deps.query,
    highRiskRules: loadPrompt('shared/high-risk-rules'),
  })

  const llm = createChatOpenAI(MODEL_STRONG).bindTools(deps.tools)
  const history: BaseMessage[] = [new SystemMessage(system), new HumanMessage(`请完成阶段目标：${stage.goal}`)]
  let consecutiveFailures = 0
  let logStep = 0

  for (let round = 1; round <= MAX_TOOL_ROUNDS; round++) {
    const resp = await llm.invoke(history)
    const reasoning = typeof resp.content === 'string' ? resp.content : ''
    if (reasoning.trim()) {
      events.push({ type: 'thought', content: reasoning.trim() })
    }

    if (!resp.tool_calls?.length) {
      return { events, outcome: reasoning.trim() || '阶段完成' }
    }

    history.push(resp)
    for (const tc of resp.tool_calls as ToolCall[]) {
      const tool = toolMap.get(tc.name)
      const argsStr = typeof tc.args === 'string' ? tc.args : JSON.stringify(tc.args)
      events.push({ type: 'action', tool_name: tc.name, content: argsStr, call_id: tc.id })

      let output: string
      const start = Date.now()
      if (!tool) {
        output = `Tool error: 未知工具 ${tc.name}`
      } else {
        try {
          const result = await tool.invoke(tc.args as Record<string, unknown>)
          output = typeof result === 'string' ? result : JSON.stringify(result)
        } catch (err) {
          output = `Tool error: ${err instanceof Error ? err.message : String(err)}`
        }
      }
      const success = !isToolFailure(output)
      const durationMs = Date.now() - start
      const truncated = output.slice(0, OBSERVATION_TRUNCATE)

      logStep++
      logToolCall({
        conversationId: deps.conversationId,
        userId: deps.userId,
        step: logStep,
        toolName: tc.name,
        inputPreview: argsStr.slice(0, 100),
        outputLength: output.length,
        durationMs,
        success,
      })
      events.push({
        type: 'observation', tool_name: tc.name, content: truncated,
        call_id: tc.id, duration_ms: durationMs, success,
      })
      history.push(new ToolMessage({ content: truncated, tool_call_id: tc.id }))
      consecutiveFailures = success ? 0 : consecutiveFailures + 1
    }

    if (consecutiveFailures >= MAX_CONSECUTIVE_FAILURES) break
  }

  // 轮次上限 / 连续失败：基于已有观察小结一次
  const summaryLlm = createChatOpenAI(MODEL_STRONG)
  const summary = await summaryLlm.invoke([
    ...history,
    new HumanMessage('已达到工具调用轮次上限，请基于以上观察输出本阶段结论（具体的事实与数据，不要复述过程）。'),
  ])
  const summaryText = typeof summary.content === 'string' ? summary.content.trim() : ''
  return { events, outcome: summaryText || '未能获得有效结论' }
}
```

注意：`resp.content` 可能是块数组（部分模型），取 string 分支；空串时兜底文案。

```typescript
// server/src/services/pne/run.ts
import type { AgentEvent, PlanStageSnapshot } from '../../types'
import type { Complexity } from '../llm-config'
import { wrapAllTools } from '../tool-adapter'
import { tools as allTools, lcTools } from '../../tools'
import { logAgentError } from '../logger'
import { planQuery } from './planner'
import { executeStage } from './executor'
import { evaluateProgress, type PneDecision } from './evaluator'
import { synthesizeAnswer } from './synthesizer'
import type { ExecutionPlan, Stage, StageResult, StageStatus, PneChatMessage } from './types'

export const MAX_REPLAN = 2

export interface PneOptions {
  systemPrompt?: string
  complexity?: Complexity
  userId?: string
  conversationId?: string
}

/** 拓扑调度：依赖全部 done 且自身 pending 的阶段为 ready 批次 */
export function findReadyStages(plan: ExecutionPlan, statuses: Map<number, StageStatus>): Stage[] {
  return plan.stages.filter(
    s =>
      (statuses.get(s.index) ?? 'pending') === 'pending' &&
      s.dependsOn.every(d => statuses.get(d) === 'done')
  )
}

function snapshot(
  plan: ExecutionPlan,
  statuses: Map<number, StageStatus>,
  stageResults: StageResult[]
): PlanStageSnapshot[] {
  return plan.stages.map(s => {
    const result = stageResults.find(r => r.index === s.index)
    return {
      index: s.index,
      goal: s.goal,
      strategy: s.strategy,
      status: statuses.get(s.index) ?? 'pending',
      dependsOn: s.dependsOn,
      outcome: result?.outcome,
    }
  })
}

function getLastUserQuery(messages: PneChatMessage[]): string {
  return [...messages].reverse().find(m => m.role === 'user')?.content || ''
}

/**
 * Plan-and-Execute 主编排。
 * 任何异常向上抛出，由 runComplex 捕获后回退 ReAct。
 */
export async function* runPneAgent(
  messages: PneChatMessage[],
  options: PneOptions
): AsyncGenerator<AgentEvent> {
  const query = getLastUserQuery(messages)
  const lcTools = wrapAllTools(allTools, lcTools)
  const toolList = lcTools.map(t => `- ${t.name} - ${t.description}`).join('\n')
  const logCtx = { conversationId: options.conversationId, userId: options.userId }

  const stageResults: StageResult[] = []
  let replanCount = 0

  try {
    while (true) {
      const plan = yield* planQuery(query, stageResults, messages, toolList)
      const statuses = new Map<number, StageStatus>(plan.stages.map(s => [s.index, 'pending' as const]))
      yield { type: 'plan', stages: snapshot(plan, statuses, stageResults) }

      let decision: PneDecision = 'continue'
      let deadlock = false

      while (true) {
        const ready = findReadyStages(plan, statuses)
        if (ready.length === 0) {
          const pending = plan.stages.filter(s => statuses.get(s.index) === 'pending')
          if (pending.length > 0) {
            deadlock = true
            for (const s of pending) statuses.set(s.index, 'skipped')
            yield { type: 'plan', stages: snapshot(plan, statuses, stageResults) }
            yield { type: 'thought', content: `【计划】${pending.length} 个阶段存在未满足的依赖，跳过` }
          }
          break
        }

        for (const s of ready) statuses.set(s.index, 'running')
        yield { type: 'plan', stages: snapshot(plan, statuses, stageResults) }

        // 无依赖阶段同批并行；事件缓冲到批次结束统一回放
        const batch = await Promise.all(
          ready.map(stage =>
            executeStage(stage, {
              tools: lcTools,
              stageResults,
              query,
              userId: options.userId,
              conversationId: options.conversationId,
            })
          )
        )
        for (const r of batch) {
          for (const ev of r.events) yield ev
        }
        for (let i = 0; i < ready.length; i++) {
          statuses.set(ready[i].index, 'done')
          stageResults.push({ index: ready[i].index, goal: ready[i].goal, outcome: batch[i].outcome })
        }
        yield { type: 'plan', stages: snapshot(plan, statuses, stageResults) }

        const evaluation = await evaluateProgress(query, plan, stageResults)
        yield { type: 'evaluation', decision: evaluation.decision, reason: evaluation.reason }
        decision = evaluation.decision

        if (decision !== 'continue') break
      }

      if (deadlock || decision === 'achieved' || decision === 'stuck') {
        if (decision === 'achieved') {
          for (const s of plan.stages) {
            if (statuses.get(s.index) === 'pending') statuses.set(s.index, 'skipped')
          }
          yield { type: 'plan', stages: snapshot(plan, statuses, stageResults) }
        }
        break
      }

      // decision === 'replan'
      replanCount++
      if (replanCount > MAX_REPLAN) {
        yield { type: 'thought', content: `【计划】replan 已达上限 ${MAX_REPLAN} 次，基于已有结果汇总` }
        break
      }
    }

    yield* synthesizeAnswer(query, stageResults, messages, options.systemPrompt)
  } catch (err) {
    logAgentError({
      ...logCtx,
      message: err instanceof Error ? err.message : String(err),
      stack: err instanceof Error ? err.stack : undefined,
    })
    throw err
  }
}
```

```typescript
          timestamp: new Date().toISOString(),
        })
        msg.thought_steps = [...thoughtSteps]
        break

      case 'evaluation':
        thoughtSteps.push({
          type: 'thought', content: `【计划评估】${event.decision}: ${event.reason}`, tool_name: null,
          timestamp: new Date().toISOString(),
        })
        msg.thought_steps = [...thoughtSteps]
        break
```
